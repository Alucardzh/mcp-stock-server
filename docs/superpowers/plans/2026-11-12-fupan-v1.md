**Goal:** 交付 v2 需求的 F1-a/F1-b/fupan_v1 一站式复盘 + R1 scale 补丁 + §7 契约扩展。

**Architecture:** klines 走 gtimg 不复权批量（8 并发）→ 指标摘要；LHB 复用现有榜单/席位加指纹去重三分类聚合层；fupan_v1 用 review.py 的 ThreadPoolExecutor 模式组装十段（每段独立降级），与 get_daily_review 共用底层 section 函数；全部经现有 `push2_client.CACHE`/`CachedData` 缓存。

**Tech Stack:** Python 3.12 / FastMCP 2.14 / pandas / requests / pytest。无新依赖。

**设计文档:** `docs/superpowers/specs/2026-11-12-fupan-v1-design.md`（唯一需求来源）

## Global Constraints

- 无新依赖；测试命令 `uv run pytest ...`（默认 `-m 'not live'`）；中文 docstring/注释；commit 前 `uv run black <files>`。
- 错误契约（§7）：`NOT_TRADE_DAY` / `NOT_READY(expected_at)` / `INVALID_ARG` 均为 `success:false` + `error_code`；条目级缺失 → 该条 `null` + 顶层 `warnings[]`，**不得静默丢**；**任何情况不得用 0 或空对象冒充取不到**。
- 段级降级：fupan_v1 某段失败 → 该段 `{"status":"unavailable","reason":...}`，其余段照常。
- 缩放：一切价格/点位输出为已还原值（板块点位即上游原值，spec 已确认）；klines 不复权。
- 指标口径：`pct60=(close/close[-60]-1)*100`；`chg5/chg10=(close/close[-5|-10]-1)*100`（bars 不足 null）；`high/low`=当日；`ma5/10/20`=不复权收盘均线；`next_limit_up/down=close*(1±limit_pct)`；`limit_pct`：60/00→0.10、30/68→0.20、北交(8/4/92开头)→0.30、名称含 ST→0.05（北交 ST 保持 0.30）。
- 现有工具返回结构不动，除：`get_board_quote` 加 `current/scale`（加法）、`breadth_section` 加 `zhaban_rate`（加法）、`get_daily_review` 顶层加 `deprecated`（加法）。
- fupan_v1 **v1 冻结**：加字段/改口径必须走 fupan_v2。
- 缓存：取数失败不写缓存；single-flight 复用 `push2_client.CACHE`。

def json_err_code(code: str, msg: str, **extra) -> str:
    """业务错误（客户端不可重试类），带机器可读 error_code；extra 顶层并入。"""
    payload = {"success": False, "error": msg, "error_code": code}
    payload.update(extra)
    return json.dumps(payload, ensure_ascii=False, indent=2)


def json_not_ready(expected_at: str) -> str:
    """整日数据未就绪（如收盘前调 fupan_v1）：客户端应稍后重试。"""
    return json.dumps(
        {
            "success": False,
            "data": None,
            "error": f"当日数据未就绪，预计 {expected_at} 后可用",
            "error_code": "NOT_READY",
            "expected_at": expected_at,
        },
        ensure_ascii=False,
        indent=2,
    )
```

def _scale_block(openq: dict) -> dict | None:
    """缩放核对块：fltt=2 值已格式化，raw = round(value × 10^f59) 无损重建。"""
    f59 = openq.get("f59")
    open_, prev = openq.get("open"), openq.get("prev_close")
    if f59 is None or open_ is None or prev is None:
        return None
    divisor = 10 ** int(f59)
    return {
        "raw_f46": int(round(open_ * divisor)),
        "raw_f60": int(round(prev * divisor)),
        "divisor": divisor,
        "f59": int(f59),
    }
```

#!/usr/bin/env python3
"""腾讯日K（web.ifzq.gtimg.cn，不复权）——fupan_v1 klines 段数据源。

- param 末段为空 = 不复权（响应键 "day"；qfq 时为 "qfqday"）；
- gtimg 非东财限流族，不经 push2 pacing，仅复用 Push2Cache 做缓存；
- 历史 end_date 数据定稿 → TTL 86400；当日 bar 盘中会变 → TTL 120
  （fupan_v1 的 NOT_READY 闸门在组装层，本层不管盘中语义）。
"""

from datetime import datetime, timedelta
import logging

import pandas as pd
import requests

from .push2_client import CACHE, EmResult, now_iso

logger = logging.getLogger(__name__)

KLINE_SESSION = requests.Session()
_KLINE_URL = "https://web.ifzq.gtimg.cn/appstock/app/fqkline/get"
_TODAY_TTL = 120
_HISTORY_TTL = 86400


def MKT_PREFIX(symbol: str) -> str:
    """6 位代码 → 腾讯市场前缀（sh/sz）。北交 8/4/92 开头返回 'bj'（支持性以探测为准）。"""
    if symbol.startswith(("6", "68", "9")):
        return f"sh{symbol}"
    if symbol.startswith(("8", "4")):
        return f"bj{symbol}"
    return f"sz{symbol}"


def _ttl(end_date: str) -> int:
    return _TODAY_TTL if end_date == str(datetime.now().date()) else _HISTORY_TTL


def fetch_kline(symbol: str, end_date: str, bars: int) -> EmResult:
    """单票不复权日K（缓存；data = DataFrame[日期,开盘,收盘,最高,最低,成交量] 升序）。"""
    key = f"kline:{symbol}:{end_date}:{bars}"

    def _fetch() -> EmResult:
        sym = MKT_PREFIX(symbol)
        start = (datetime.strptime(end_date, "%Y-%m-%d") - timedelta(days=int(bars * 2))).strftime("%Y-%m-%d")
        try:
            r = KLINE_SESSION.get(
                _KLINE_URL,
                params={"param": f"{sym},day,{start},{end_date},{bars},"},
                timeout=8,
            )
            payload = r.json()
        except Exception as e:  # noqa: BLE001
            logger.warning("gtimg kline %s failed: %s", symbol, e)
            return EmResult(ok=False, reason="upstream_error")
        node = (payload.get("data") or {}).get(sym) or {}
        rows = node.get("day") or node.get("qfqday") or []
        if not rows:
            return EmResult(ok=False, reason="upstream_error")
        df = pd.DataFrame(
            [[r[0], float(r[1]), float(r[2]), float(r[3]), float(r[4]), float(r[5])] for r in rows],
            columns=["日期", "开盘", "收盘", "最高", "最低", "成交量"],
        )
        return EmResult(ok=True, data=df, as_of=now_iso())

    return CACHE.get(key, _ttl(end_date), _fetch)
```

#!/usr/bin/env python3
"""个股日K指标摘要（不复权）：MA/区间涨幅/涨跌停价——fupan_v1 klines 段口径。

口径（spec §4.2）：ma5/10/20=不复权收盘均线；pct60=(close/close[-60]-1)*100；
chg5/chg10=(close/close[-5|-10]-1)*100，bars 不足为 None；high/low=当日；
next_limit_up/down=close*(1±limit_pct)。
"""

from concurrent.futures import ThreadPoolExecutor
import logging

import pandas as pd

from .gtimg_kline import fetch_kline  # noqa: F401  (测试 monkeypatch 点)

logger = logging.getLogger(__name__)

_BATCH_WORKERS = 8


def limit_pct(symbol: str, name: str = "") -> float:
    """涨跌停幅度：北交 0.30（含北交ST）> ST 0.05 > 科创创业 0.20 > 主板 0.10。"""
    if symbol.startswith(("8", "4", "92")):
        return 0.30
    if "ST" in (name or "").upper():
        return 0.05
    if symbol.startswith(("30", "68")):
        return 0.20
    return 0.10


def _ma(closes: list[float], n: int):
    if len(closes) < n:
        return None
    return round(sum(closes[-n:]) / n, 3)


def _chg(closes: list[float], n: int):
    if len(closes) <= n or not closes[-n - 1]:
        return None
    return round((closes[-1] / closes[-n - 1] - 1) * 100, 3)


def compute_metrics(df: pd.DataFrame, name: str = "") -> dict:
    """DataFrame（gtimg_kline 列名）→ 指标 dict。bars 不足的指标为 None。

    symbol/name 优先取显式入参，其次 df.attrs（fetch_kline 写入，只读）。
    """
    closes = [float(x) for x in df["收盘"].tolist()]
    last = df.iloc[-1]
    symbol = str(df.attrs.get("symbol", ""))
    name = name or str(df.attrs.get("name", ""))
    lp = limit_pct(symbol, name)
    close = closes[-1]
    return {
        "close": close,
        "ma5": _ma(closes, 5),
        "ma10": _ma(closes, 10),
        "ma20": _ma(closes, 20),
        "pct60": _chg(closes, 60),
        "chg5": _chg(closes, 5),
        "chg10": _chg(closes, 10),
        "high": float(last["最高"]),
        "low": float(last["最低"]),
        "limit_pct": lp,
        "next_limit_up": round(close * (1 + lp), 3),
        "next_limit_down": round(close * (1 - lp), 3),
    }
```

def _one(symbol: str, end_date: str, bars: int):
    res = fetch_kline(symbol, end_date, bars)
    if not res.ok or res.data is None or res.data.empty:
        return symbol, None
    return symbol, res.data  # attrs（symbol/name）由 fetch_kline 写入，此处只读


def batch_klines(symbols: list[str], end_date: str, bars: int = 130) -> dict:
    """并发取数并算指标：{items: {code: dict|None}, warnings: [...]}（无 JSON 信封）。"""
    symbols = sorted({s.strip() for s in symbols if s and s.strip()})
    with ThreadPoolExecutor(max_workers=_BATCH_WORKERS) as pool:
        results = list(pool.map(lambda s: _one(s, end_date, bars), symbols))
    items, warnings = {}, []
    for symbol, df in results:
        if df is None:
            items[symbol] = None
            warnings.append(f"{symbol} 日K获取失败")
            continue
        items[symbol] = compute_metrics(df)
    return {"items": items, "warnings": warnings}

#!/usr/bin/env python3
"""龙虎榜聚合口径（F1-b）：榜净并集 / 席位指纹去重 / 三分类净额（万元）。

口径（spec §5.1，v2 §6.4）：
- net = **max-buy 记录的 买−卖**（行一致口径；真实数据同票多理由上榜金额一致，等价于 v2 字面的 max买−max卖；F6 实盘验收最终校验）；
- 席位指纹去重：合并键 = 席位名+买额+卖额（东财偶发整行重复）；
- 三分类 = 去重后席位行净额（买-卖）按 [机构]/[外资]/[游资] 汇总；
- diff = net - seat_net（榜面 vs 席位前5，口径不同不必然 0，参考值）。
"""

from datetime import datetime
import logging

from akshare import stock_lhb_stock_detail_em

from .lhb import _classify_seat, _fetch_billboard, _norm_billboard
from .push2_client import CACHE, now_iso
from .review_common import safe_num

logger = logging.getLogger(__name__)

_EIGHTEEN = 18  # 当日 18:00 前席位未定稿


def _ttl(day_str: str) -> int:
    if day_str < str(datetime.now().date()):
        return 86400
    return 300 if datetime.now().hour < _EIGHTEEN else 86400


def _seat_rows(symbol: str, d_compact: str):
    """买/卖两侧原始席位行 [(name, buy_wan|None, sell_wan|None)]，失败侧跳过。"""
    rows = []
    for flag, idx in (("买入", 1), ("卖出", 2)):
        try:
            df = stock_lhb_stock_detail_em(symbol=symbol, date=d_compact, flag=flag)
        except Exception as e:  # noqa: BLE001
            logger.warning("seats %s %s failed: %s", symbol, flag, e)
            continue
        if df is None or df.empty:
            continue
        exact = "买入金额" if flag == "买入" else "卖出金额"
        amt_col = exact if exact in df.columns else next(
            (c for c in df.columns if "金额" in str(c)), None)
        name_col = "营业部名称" if "营业部名称" in df.columns else next(
            (c for c in df.columns if "营业部" in str(c)
             or "OPERATEDEPT" in str(c).upper()), None)
        if amt_col is None or name_col is None:
            continue
        for _, r in df.head(5).iterrows():
            key = (idx == 1, str(r[name_col] or "?"),
                   safe_num(r[amt_col] / 1e4, 4) if r[amt_col] is not None else None)
            rows.append((key[1], key[2] if key[0] else None, None if key[0] else key[2]))
    return rows

```python
def _aggregate_stocks(day) -> tuple[dict, list[str]]:
    """榜单 → 按股聚合（万元）。返回 (stocks, notes)。"""
    from .review_common import parse_day  # noqa: F401  day 已是 date

    df = _fetch_billboard(day)
    if df is None or df.empty:
        return {}, [f"{day} 无龙虎榜数据(可能非交易日或榜单未出)"]
    raw = _norm_billboard(df)
    by_code: dict[str, dict] = {}
    for r in raw:
        a = by_code.setdefault(r["code"], {"name": r["name"], "recs": [], "reasons": []})
        a["recs"].append((safe_num(r["buy"], 4) or 0.0, safe_num(r["sell"], 4) or 0.0))
        if r["reason"] and r["reason"] not in a["reasons"]:
            a["reasons"].append(r["reason"])
    d_compact = day.strftime("%Y%m%d")
    stocks = {}
    for code, a in by_code.items():
        top = max(a["recs"], key=lambda p: p[0])  # max-buy 记录（行一致口径，见接口说明）
        net = (top[0] - top[1]) / 1e4  # 元 → 万元
        rows = _seat_rows(code, d_compact)
        seen, inst, foreign, hot, buy_side_n, sell_side_n = set(), 0.0, 0.0, 0.0, 0, 0
        seat_net = 0.0
        for name, buy_wan, sell_wan in rows:
            fp = (name, buy_wan, sell_wan)
            if fp in seen:  # 指纹去重
                continue
            seen.add(fp)
            if buy_wan is not None:
                buy_side_n += 1
            else:
                sell_side_n += 1
            amt = (buy_wan or 0.0) - (sell_wan or 0.0)
            seat_net += amt
            cls = _classify_seat(name)
            if cls == "[机构]":
                inst += amt
            elif cls == "[外资]":
                foreign += amt
            else:
                hot += amt
        stocks[code] = {
            "name": a["name"],
            "net": safe_num(net, 4),
            "buy": safe_num(top[0] / 1e4, 4),
            "sell": safe_num(top[1] / 1e4, 4),
            "inst": safe_num(inst, 4),
            "foreign": safe_num(foreign, 4),
            "hot": safe_num(hot, 4),
            "seat_n": buy_side_n + sell_side_n,
            "seat_net": safe_num(seat_net, 4),
            "diff": safe_num(net - seat_net, 4),
            "reasons": a["reasons"],
        }
    return stocks, []


def lhb_aggregate(date: str, symbols: list[str] = []) -> dict:
    """聚合入口（缓存；无 JSON 信封）。symbols 为空 → 全部上榜票。"""
    from .review_common import parse_day

    day = parse_day(date)
    day_str = str(day)

    def _fetch() -> "EmResult":  # noqa: F821
        from .push2_client import EmResult

        stocks, notes = _aggregate_stocks(day)
        if not stocks:
            return EmResult(ok=False, reason="upstream_error")  # 空榜单不缓存
        return EmResult(ok=True, data={"stocks": stocks, "notes": notes},
                        as_of=now_iso())

    res = CACHE.get(f"lhb_agg:{day_str}", _ttl(day_str), _fetch)
    stocks = (res.data or {}).get("stocks", {}) if res.ok else {}
    notes = (res.data or {}).get("notes", []) if res.ok else [f"龙虎榜聚合失败: {res.reason}"]
    if symbols:
        wanted = {s.strip() for s in symbols}
        stocks = {c: v for c, v in stocks.items() if c in wanted}
    cats = {
        "inst": safe_num(sum(v["inst"] or 0 for v in stocks.values()), 4),
        "foreign": safe_num(sum(v["foreign"] or 0 for v in stocks.values()), 4),
        "hot": safe_num(sum(v["hot"] or 0 for v in stocks.values()), 4),
    }
    total_net = safe_num(sum(v["net"] or 0 for v in stocks.values()), 4) if stocks else None
    return {
        "date": day_str, "stocks": stocks, "cats": cats,
        "total_net": total_net, "notes": notes, "as_of": res.as_of if res.ok else "",
        "cached": res.cached, "stale": res.stale, "source": "eastmoney_datacenter",

def pool_full_snapshot(pool: str, day: date_type) -> dict | None:
    """严格日期（不回退）三池 full 行快照，键名映射为 v2 英文；空池/失败 None。

    fupan_v1 专用：NOT_TRADE_DAY 判定与段级降级在组装层。
    """
    df, as_of, cached, stale = _pool_cached(pool, day)
    if df is None or df.empty:
        return None
    fields = [c for c in _POOL_FIELDS.get(pool, []) if c in df.columns]
    items = []
    for _, r in df.iterrows():
        row = {}
        for c in fields:
            v = r[c]
            if c in MONEY_FIELDS:
                v = safe_num(float(v) / 1e8, 2)
            elif c in ("涨跌幅", "换手率"):
                v = safe_num(v, 2)
            row[_POOL_KEY_MAP.get(c, c)] = v
        items.append(row)
    return {
        "pool": pool, "date": str(day), "count": len(df), "truncated": False,
        "items": items, "as_of": as_of, "cached": cached, "stale": stale,
        "source": "eastmoney_push2ex", "notes": [],
    }


def industry_counts(pool: str, day: date_type) -> dict | None:
    """三池按所属行业分桶计数（通用版；行业名为东财 4 字截断原文）。"""
    try:
        df = _pool_cached(pool, day)[0]
        if df is None or df.empty or "所属行业" not in df.columns:
            return None
        return {str(k): int(v) for k, v in df["所属行业"].value_counts().items()}
    except Exception:  # noqa: BLE001
        return None

def fupan_boards_section() -> dict | None:
    """fupan_v1 boards 段：行业+概念全量行（无 limit 截断），open_pct 为 clist 比值。

    行 = _emit_row 形态 + open/prev_close(常为 None) + open_pct。
    单 scope 失败 → 该 scope 为 {"status":"unavailable","reason"}；双失败 → None。
    """
    snaps = {"industry": fetch_board_list("行业"), "concept": fetch_board_list("概念")}
    if not any(s.ok for s in snaps.values()):
        return None
    counts = zt_industry_counts()
    out = {"as_of": "", "stale": False, "cached": True, "source": _SOURCE}
    for scope, s in snaps.items():
        if not s.ok:
            out[scope] = {"status": "unavailable", "reason": s.reason or "upstream_error"}
            continue
        boards = [dict(b) for b in s.data["boards"]]  # 缓存引用共享，先拷贝
        for b in boards:
            b["open"] = b["_open"]
            b["prev_close"] = b["_prev_close"]
            b["open_pct"] = _open_pct(b["_open"], b["_prev_close"])
        out[scope] = [_emit_row(b, counts) for b in boards]
        out["as_of"] = max(out["as_of"], s.as_of)
        out["stale"] = out["stale"] or s.stale
        out["cached"] = out["cached"] and s.cached
    return out

def _is_not_ready(day: date_type) -> bool:
    if day != datetime.now().date():
        return False
    now = datetime.now()
    return (now.hour, now.minute) < (_READY_HOUR, _READY_MINUTE)


def _chain_section(chain_buckets: dict, zt_items: list, zb_items: list) -> dict:
    """链强度：{链名: {buckets, limitup, zhaban, total, rate, members}}。"""
    out = {}
    for chain, buckets in (chain_buckets or {}).items():
        bset = set(buckets)
        zt = [i for i in zt_items if i.get("hybk") in bset]
        zb = [i for i in zb_items if i.get("hybk") in bset]
        limitup, zhaban = len(zt), len(zb)
        total = limitup + zhaban
        out[chain] = {
            "buckets": sorted(bset),
            "limitup": limitup,
            "zhaban": zhaban,
            "total": total,
            "rate": round(zhaban / total * 100, 2) if total else 0.0,
            "members": [{"code": i.get("code"), "name": i.get("name"),
                         "lianban": i.get("lianban")} for i in zt],
        }
    return out


def _bucket_section(day: date_type) -> dict:
    """每个 hybk 桶的涨停/炸板/合计/炸板率（涨停+炸板分桶合并）。"""
    zt = industry_counts("涨停", day) or {}
    zb = industry_counts("炸板", day) or {}
    out = {}
    for hybk in set(zt) | set(zb):
        limitup, zhaban = zt.get(hybk, 0), zb.get(hybk, 0)
        total = limitup + zhaban
        out[hybk] = {"limitup": limitup, "zhaban": zhaban, "total": total,
                     "rate": round(zhaban / total * 100, 2) if total else 0.0}
    return out

    """fupan_v1：晚班复盘一站式（指数/宽度/资金流/三池/桶/板块/链/个股指标/龙虎榜/衍生品）

    Args:
        date: 复盘日 YYYY-MM-DD（收盘口径；非交易日报 NOT_TRADE_DAY）
        symbols: 需要个股指标的代码（与当日涨停池全量合并去重）
        options: with_* 开关 / chain_buckets {链名:[hybk...]} / kline_bars(默认130)
    """
    try:
        day = parse_day(date)
        if day is None:
            return json_err_code("INVALID_ARG", f"日期格式错误: {date}，请使用 YYYY-MM-DD")
        if day > datetime.now().date():
            return json_err_code("INVALID_ARG", f"查询日期 {day} 晚于今天")
        if _is_not_ready(day):
            return json_not_ready("15:05")

        opts = {**_DEFAULT_OPTIONS, **{k: v for k, v in (options or {}).items()
                                       if k in _DEFAULT_OPTIONS or k in ("chain_buckets", "kline_bars")}}
        key = f"{day}|{','.join(sorted(symbols))}|{json.dumps(opts, sort_keys=True, ensure_ascii=False)}"
        cached = _fupan_cache.get(key)
        if cached is not None and not cached.is_expired():
            payload = json.loads(cached.data)
            payload["data"]["cached"] = True
            return json.dumps(payload, ensure_ascii=False, indent=2)

        # 先取涨停池（klines 默认集 + NOT_TRADE_DAY 判定都要它）
        zt_snap = pool_full_snapshot("涨停", day)
        zb_snap = pool_full_snapshot("炸板", day) if opts["with_pools"] else None
        try:
            indices = indices_section(day)
        except Exception:  # noqa: BLE001
            indices = None
        if zt_snap is None and (zb_snap is None or not opts["with_pools"]) and indices is None:
            return json_err_code("NOT_TRADE_DAY", f"{day} 无涨停池与指数数据(可能非交易日)")

        bad = {str(s) for s in (symbols or []) if not (str(s).isdigit() and len(str(s)) == 6)}
        warnings = [f"代码格式异常: {s}" for s in sorted(bad)]
        kline_symbols = sorted((set(map(str, symbols or [])) - bad) | {
            i["code"] for i in (zt_snap or {}).get("items", []) if i.get("code")})

        def run(name, fn):
            try:
                val = fn()
                return name, val if not isinstance(val, str) else json.loads(val), None
            except Exception as e:  # noqa: BLE001
                logger.warning("fupan section %s failed: %s", name, e)
                return name, None, str(e)

        jobs = []
        if opts["with_breadth"]:
            jobs.append(("breadth", lambda: breadth_section(day)))
        if opts["with_fund_flow"]:
            jobs.append(("fund_flow", lambda: fund_flow_section(day)))
        if opts["with_derivatives"]:
            jobs.append(("derivatives", lambda: derivatives_section(day)))
        if opts["with_boards"]:
            jobs.append(("boards", fupan_boards_section))
        if opts["with_lhb"]:
            jobs.append(("lhb", lambda: lhb_aggregate(str(day), [])))
        if opts["with_klines"]:
            bars = int(opts.get("kline_bars", 130))
            jobs.append(("klines", lambda: batch_klines(kline_symbols, str(day), bars)))
        with ThreadPoolExecutor(max_workers=6) as pool:
            results = list(pool.map(lambda nf: run(nf[0], nf[1]), jobs))

        sections = {
            "indices": indices,
            "pools": (
                {"zt": zt_snap, "zb": zb_snap,
                 "prev_zt": pool_full_snapshot("昨涨停", day),
                 "dt": pool_full_snapshot("跌停", day)}
                if opts["with_pools"] else {"status": "skipped", "reason": "with_pools=false"}
            ),
            "buckets": _bucket_section(day) if opts["with_pools"] else {},
            "chains": _chain_section(opts.get("chain_buckets") or {},
                                     (zt_snap or {}).get("items", []),
                                     (zb_snap or {}).get("items", [])),
        }
        any_stale = bool(zt_snap and zt_snap.get("stale")) or bool(zb_snap and zb_snap.get("stale"))
        any_cached = bool(zt_snap and zt_snap.get("cached")) and (zb_snap is None or zb_snap.get("cached"))
        for name, val, err in results:
            if err is not None or val is None:
                sections[name] = {"status": "unavailable",
                                  "reason": (err or "upstream_error")[:120]}
                continue
            if name == "klines":  # {code: 指标} 直出（v2 §6.3），warnings 上浮
                warnings.extend(val.get("warnings", []))
                val = val.get("items", {})
            sections[name] = val
            if isinstance(val, dict):
                any_stale = any_stale or bool(val.get("stale"))
                any_cached = any_cached and bool(val.get("cached", True))
        # lhb/dt 特例：跌停池 >30 天上游限制 → note
        if opts["with_pools"] and sections["pools"].get("dt") is None and day < datetime.now().date():
            sections["pools"]["dt"] = {"status": "unavailable",
                                       "reason": "dt_pool_30d_limit"}  # 探测后如确认则保留

        data = {
            "schema_version": "v1",
            "date": str(day),
            "as_of": now_iso(),
            "stale": any_stale,
            "cached": False,
            "sections": sections,
        }
        if warnings:
            data["warnings"] = warnings
        out = json_ok(data)
        _fupan_cache[key] = CachedData(out, ttl=300)
        return out
    except Exception as e:  # noqa: BLE001
        logger.error("Error in fupan_v1: %s", e)
        return json_err(f"fupan_v1 失败: {e}")
```
